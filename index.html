// --- FİŞ OKUMA (OCR) VE DOLAP İŞLEMLERİ (Gelişmiş Filtreli) --- async function fisOku() { const fileInput = document.getElementById('fisInput'); const status = document.getElementById('ocrStatus'); const btn = document.getElementById('btnOcr'); if (!fileInput.files || fileInput.files.length === 0) return alert("Lütfen fiş görseli seçin!"); btn.disabled = true; status.innerText = "⏳ Yapay zeka fişi tarıyor ve gereksiz bilgileri ayıklıyor..."; try { const file = fileInput.files[0]; const result = await Tesseract.recognize(file, 'tur'); const tumSatirlar = result.data.text.split('\n'); // Yasaklı kelimeler listesi const engellenenler = [ 'POŞET', 'POS ET', 'TARİH', 'TARIH', 'SAAT', 'FİŞ NO', 'FIS NO', 'TOPKDV', 'TOPLAM', 'MARKET', 'TİC', 'VERGİ', 'V.D', 'ANADOLU', 'KDV', 'NAKİT', 'KREDİ', 'PARA', 'PARA ÜSTÜ', 'TL', 'SIRA NO' ]; const filtrelenmisUrunler = tumSatirlar.map(s => s.trim()).filter(satir => { if (satir.length < 3) return false; const ust = satir.toUpperCase(); // Yasaklı kelimelerden biri geçiyorsa atla if (engellenenler.some(kelime => ust.includes(kelime))) return false; // Sadece tarih/saat veya sayı içeren satırları atla if (/^\d{2}[\/\.]\d{2}[\/\.]\d{4}/.test(satir)) return false; if (/^[\d\s\.,:\-]+$/.test(satir)) return false; return true; }).map(urun => { // Ürün adının sonundaki fiyat, miktar ve tuhaf simgeleri temizle return urun .replace(/[\d\.,"'\-–—&*]+\s*$/, '') // Sonda kalan sayı/fiyatları sil .replace(/\s+x\d+.*$/i, '') // "x20 ..." gibi adet bilgilerini sil .trim(); }).filter(urun => urun.length > 2); // Temizlik sonrası çok kısa kalanları sil if (filtrelenmisUrunler.length === 0) { status.innerText = "⚠️ Fişten geçerli bir ürün adı okunamadı."; btn.disabled = false; return; } const eklenecekler = filtrelenmisUrunler.map(urun => ({ malzeme_adi: urun, ekleyen_kisi: aktifKullanici })); const { error } = await _supabase.from('dolaptakiler').insert(eklenecekler); if (error) { console.error("Supabase Ekleme Hatası:", error); status.innerText = "❌ Veritabanına eklenirken hata oluştu: " + error.message; } else { status.innerText = `✅ ${filtrelenmisUrunler.length} adet ürün temizlenerek dolaba eklendi!`; dolapGetir(); tarifleriGetir(); } } catch (e) { console.error("OCR Hatası:", e); status.innerText = "❌ Okuma hatası oluştu."; } finally { btn.disabled = false; } }
